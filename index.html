#include #include #include #include #include #define SERVICE_UUID "4fafc201-1fb5-459e-8fcc-c5c9c331914b" #define CHARACTERISTIC_UUID "beb5483e-36e1-4688-b7f5-ea07361b26a8" // Nút nhấn ngắt kết nối (mặc định dùng nút BOOT GPIO 0 trên ESP32) #define BUTTON_PIN 0 BLEServer* pServer = nullptr; BLECharacteristic* pCharacteristic = nullptr; bool deviceConnected = false; bool oldDeviceConnected = false; unsigned long lastNotifyTime = 0; const unsigned long notifyInterval = 1000; // Gửi dữ liệu mỗi 1000ms (1s) class MyServerCallbacks : public BLEServerCallbacks { void onConnect(BLEServer* pServer) { deviceConnected = true; Serial.println(">> Da co dien thoai ket noi!"); } void onDisconnect(BLEServer* pServer) { deviceConnected = false; Serial.println(">> Dien thoai ngat ket noi!"); } }; void setup() { Serial.begin(115200); delay(1000); Serial.println("\n=== TEST BLE SERVER BMS ==="); // Cấu hình nút nhấn (dùng điện trở kéo lên nội bộ) pinMode(BUTTON_PIN, INPUT_PULLUP); // Khởi tạo bộ sinh số ngẫu nhiên theo nhiễu ADC randomSeed(analogRead(34)); BLEDevice::init("ESP32_BMS_TEST"); pServer = BLEDevice::createServer(); pServer->setCallbacks(new MyServerCallbacks()); BLEService *pService = pServer->createService(SERVICE_UUID); pCharacteristic = pService->createCharacteristic( CHARACTERISTIC_UUID, BLECharacteristic::PROPERTY_READ | BLECharacteristic::PROPERTY_NOTIFY ); pCharacteristic->addDescriptor(new BLE2902()); pService->start(); BLEAdvertising *pAdvertising = BLEDevice::getAdvertising(); pAdvertising->addServiceUUID(SERVICE_UUID); pAdvertising->setScanResponse(true); pAdvertising->setMinPreferred(0x06); pAdvertising->setMaxPreferred(0x12); BLEDevice::startAdvertising(); Serial.println(">> Dang quang ba. Nhan nut BOOT de chu dong ngat ket noi BLE."); } void loop() { // 1. Kiểm tra nút bấm ngắt kết nối (Active LOW) if (digitalRead(BUTTON_PIN) == LOW) { delay(50); // Chống rung phím (debounce) if (digitalRead(BUTTON_PIN) == LOW && deviceConnected) { Serial.println(">> Nut nhan duoc bam: Chu dong ngat ket noi BLE!"); pServer->disconnect(pServer->getConnId()); // Đợi người dùng nhả nút while (digitalRead(BUTTON_PIN) == LOW) { delay(10); } } } // 2. Tạo và gửi dữ liệu ngẫu nhiên mỗi 1 giây khi có kết nối if (deviceConnected) { unsigned long currentMillis = millis(); if (currentMillis - lastNotifyTime >= notifyInterval) { lastNotifyTime = currentMillis; // Điện áp: 48.0V - 54.5V float v = random(480, 546) / 10.0; // Dòng điện: 2.0A - 25.0A float i = random(20, 251) / 10.0; // Phần trăm pin (SOC): 20% - 100% int soc = random(20, 101); char payload[64]; snprintf(payload, sizeof(payload), "{\"v\":%.1f,\"i\":%.1f,\"soc\":%d}", v, i, soc); pCharacteristic->setValue(payload); pCharacteristic->notify(); Serial.printf("Notify: %s\n", payload); } } // 3. Xử lý bật lại quảng bá khi bị ngắt kết nối if (!deviceConnected && oldDeviceConnected) { delay(500); // Chờ stack BLE giải phóng bộ nhớ pServer->startAdvertising(); Serial.println(">> Bat lai quang ba BLE..."); oldDeviceConnected = deviceConnected; } // 4. Xử lý trạng thái vừa kết nối if (deviceConnected && !oldDeviceConnected) { oldDeviceConnected = deviceConnected; } }
